SELECT DISTINCT(UF) FROM PESSOA ORDER BY UF;
SELECT COUNT(UF)FROM PESSOA;

SELECT   CASE
   WHEN (UF IN ('RS', 'SC', 'PR')) THEN
        'SUL' 
   WHEN (UF IN ('SP','RJ', 'ES', 'MG')) THEN 
        'SUDESTE'
   WHEN (UF IN ('MT','MS','GO','DF')) THEN 
        'CENTRO-OESTE' 
   WHEN (UF IN ('AC','RO','AM','RR','AP','PA')) THEN
        'NORTE'
   WHEN (UF IN ('MA','PI','CE','RN','PB','PE','AL', 'SE','BA')) THEN
        'NORDESTE'
  END AS REGIAO , UF, COUNT (*) AS TOTAL_CLIENTES
FROM PESSOA
GROUP BY UF;
